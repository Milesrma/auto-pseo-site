function setBrand(full) {

var parts = String(full).split('|');

var name = parts[0].trim() || CONFIG.brand;

var descriptor = parts.length > 1 ? parts.slice(1).join('|').trim() : CONFIG.descriptor;


var logo = document.getElementById('brand-logo');

if (logo) logo.textContent = name;


var heroTitle = document.getElementById('hero-title');

if (heroTitle) {

var slot = heroTitle.querySelector('[data-brand-slot]');

if (slot) slot.textContent = name; else heroTitle.textContent = name;

}


var footer = document.getElementById('footer-brand');

if (footer) footer.textContent = name;


$$('.brand-mirror').forEach(function (el) { el.textContent = name; });

document.title = name + ' | ' + descriptor;

refreshWaLinks();

}


function setCity(city) {

var formatted = String(city).split(/[,|•·\/]+/).map(function (s) { return s.trim(); }).filter(Boolean).join(' • ').toUpperCase();

if (!formatted) return;

var badge = document.getElementById('brand-city-badge');

if (badge) badge.textContent = formatted;

$$('.footer-city').forEach(function (el) { el.textContent = formatted; });

}


function setTagline(text) {

var sub = document.getElementById('hero-subtitle');

if (sub && String(text).trim()) sub.textContent = String(text).trim();

}


function setPhone(phone) {

var digits = String(phone).replace(/\D/g, '');

if (digits.length < 6) return;

state.phone = digits;

refreshWaLinks();

}


function setAccent(input) {

var hex = String(input).trim().replace(/^#/, '');

if (/^[0-9a-fA-F]{3}$/.test(hex)) hex = hex.split('').map(function (c) { return c + c; }).join('');

if (!/^[0-9a-fA-F]{6}$/.test(hex)) return;

var r = parseInt(hex.slice(0, 2), 16), g = parseInt(hex.slice(2, 4), 16), b = parseInt(hex.slice(4, 6), 16);

var root = document.documentElement;

root.style.setProperty('--accent', '#' + hex);

root.style.setProperty('--accent-rgb', r + ',' + g + ',' + b);

if (three) three.setAccent('#' + hex);

}


function setRate(rate) {

var n = parseFloat(String(rate).replace(/[^0-9.]/g, ''));

if (!isFinite(n) || n <= 0) return;

var el = document.getElementById('calc-rate');

if (el) el.textContent = nf.format(n);

updateEstimate(true);

}


function setCurrency(code) {

var c = String(code).trim().toUpperCase().slice(0, 4);

if (!c) return;

state.currency = c;

updateEstimate(true);

}


function applyChameleonParams() {

var params = new URLSearchParams(window.location.search);


var brand = params.get('brand');

if (brand && brand.trim()) setBrand(brand.trim());


var city = params.get('city');

if (city && city.trim()) setCity(city);


var tagline = params.get('tagline');

if (tagline && tagline.trim()) setTagline(tagline);


var phone = params.get('phone');

if (phone && phone.trim()) setPhone(phone);


var accent = params.get('accent');

if (accent && accent.trim()) setAccent(accent);


var rate = params.get('rate');

if (rate && rate.trim()) setRate(rate);


var currency = params.get('currency');

if (currency && currency.trim()) setCurrency(currency);


updateEstimate(true);

refreshWaLinks();

}

window.CHAMELEON = {

setAccent: setAccent, setRate: setRate, setCurrency: setCurrency,

apply: applyChameleonParams, defaults: CONFIG

};


if (document.readyState === 'loading') {

}
